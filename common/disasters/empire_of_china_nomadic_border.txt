empire_of_china_nomadic_border = {
	potential = {
		has_dlc = "Mandate of Heaven"
		is_emperor_of_china = yes
	}

	can_start = {
		has_any_disaster = no
		any_neighbor_country = {
			is_subject = no
			NOT = { truce_with = ROOT }
			NOT = { alliance_with = ROOT }
			custom_trigger_tooltip = {
				tooltip = empire_of_china_nomadic_border_tt
				OR = {
					has_reform = steppe_horde
					has_reform = great_mongol_state_reform
				}
				total_own_and_non_tributary_subject_development = 300
			}
			#NOT = { is_subject_of = ROOT }
		}
	}
	
	can_stop = { 
		OR = {
			has_any_disaster = yes
			NOT = {
				any_neighbor_country = {
					NOT = { truce_with = ROOT }
					NOT = { alliance_with = ROOT }
					is_subject = no
					custom_trigger_tooltip = {
						tooltip = empire_of_china_nomadic_border_tt
						OR = {
							has_reform = steppe_horde
							has_reform = great_mongol_state_reform
							hidden_trigger = {
								has_global_flag = is_forming_the_mongol_empire
							}
						}
						total_own_and_non_tributary_subject_development = 300
					}
				}
			}
		}
		hidden_trigger = {
			NOT = { has_global_flag = is_forming_the_mongol_empire }
		}
	}
	
	
	progress = {
		modifier = {
			factor = 1
			any_neighbor_country = {
				custom_trigger_tooltip = {
					tooltip = empire_of_china_nomadic_border_tt
					OR = {
						has_reform = steppe_horde
						has_reform = great_mongol_state_reform
					}
					total_own_and_non_tributary_subject_development = 300
				}
				is_subject = no
				NOT = { truce_with = ROOT }
				NOT = { alliance_with = ROOT }
				#NOT = { is_subject_of = ROOT }
			}
		}
		modifier = {
			factor = 1.5
			any_neighbor_country = {
				custom_trigger_tooltip = {
					tooltip = empire_of_china_nomadic_border_tt
					OR = {
						has_reform = steppe_horde
						has_reform = great_mongol_state_reform
					}
					total_own_and_non_tributary_subject_development = 300
				}
				war_with = ROOT
				war_score_against = {
					who = ROOT
					value = 10
				}
			}
		}
		modifier = {
			factor = 3
			any_neighbor_country = {
				custom_trigger_tooltip = {
					tooltip = empire_of_china_nomadic_border_tt
					OR = {
						has_reform = steppe_horde
						has_reform = great_mongol_state_reform
					}
					total_own_and_non_tributary_subject_development = 300
				}
				war_with = ROOT
				war_score_against = {
					who = ROOT
					value = 25
				}
			}
		}
		modifier = {
			factor = -3
			any_neighbor_country = {
				custom_trigger_tooltip = {
					tooltip = empire_of_china_nomadic_border_tt
					OR = {
						has_reform = steppe_horde
						has_reform = great_mongol_state_reform
					}
					total_own_and_non_tributary_subject_development = 300
				}
				custom_trigger_tooltip = {
					tooltip = warscore_v_horde_25_tt
					ROOT = {
						war_score_against = {
							who = PREV
							value = 25
						}
					}
				}
			}
		}
	}
	

	
	can_end = {
		OR = {
			is_emperor_of_china = no
			NOT = {
				any_neighbor_country = {
					NOT = { alliance_with = ROOT }
					#@ new. can_stop already had this line; can_end did not, which is
					#@ why winning a war against the horde could not end the disaster.
					#@ cb_insult's peace list is po_revoke_cores / po_gold /
					#@ po_concede_defeat / po_humiliate_rival - no land and no
					#@ subjugation - so the "show superiority" option in
					#@ mht_nomadic.3 could be won outright and still leave the
					#@ disaster running forever. Any peace now clears it.
					#@
					#@ It comes back. This disaster is one of only 12 of 51 with no
					#@ recurrence guard in potential/can_start - ming_crisis and
					#@ empire_of_china_culture both have one, this does not - and
					#@ on_start_effect clears its own ticker flags precisely so it
					#@ can run again. Truce is TRUCE_YEARS 5, plus up to
					#@ SCALED_TRUCE_YEARS 10 scaled by warscore taken, so a crushing
					#@ win buys 15 years of quiet and a white peace buys 5.
					NOT = { truce_with = ROOT }
					is_subject = no
					custom_trigger_tooltip = {
						tooltip = empire_of_china_nomadic_border_tt
						OR = {
							has_reform = steppe_horde
							has_reform = great_mongol_state_reform
							hidden_trigger = {
								has_global_flag = is_forming_the_mongol_empire
							}
						}
						total_own_and_non_tributary_subject_development = 300
					}
				}
			}
		}
		hidden_trigger = {
			NOT = { has_global_flag = is_forming_the_mongol_empire }
		}
	}
	
	#@ overall I want this disaster to be less punishing and symbolizes the empire
	#@ dealing with harassments and preparing for war, which has upsides
	#@ and downsides
	modifier = {
		imperial_mandate = -0.25			#@ was -0.3, slightly toned down
		reduced_liberty_desire = -25
		global_unrest = 5					#@ new
		fort_maintenance_modifier = 0.25	#@ new
		defensiveness = 0.1					#@ a buff: signals stronger defenses
		leader_cost = -0.25					#@ a buff: war times call for more generals
		land_maintenance_modifier = 0.25	#@ new
		army_tradition = 1					#@ a buff
		#@ land_morale = -0.15
	}	
	
	
	on_start = empire_of_china_nomadic_border_events.1
	on_end = mht_nomadic.4		#@ was empire_of_china_nomadic_border_events.2; ours clears
								#@ the ultimatum flag then fires that event unchanged
	
	on_monthly = {
		events = {
			mht_nomadic.1 #@ our ultimatum chain - forces the standoff to resolve
			mch_missions.5 # Koxinga flees to Taiwan
			ming_crisis.3 # Li Zicheng's Rebellion and the Shun Dynasty
			ming_crisis.4 # Autonomy for Yunnan province
			ming_crisis.5 # Losing Control in the South
		}
		random_events = { 
			#1000 = 0
			# None
		}
	}

	on_start_effect = {
		hidden_effect = {
			clr_country_flag = empire_of_china_nomadic_border_started_ticker
			clr_country_flag = empire_of_china_nomadic_border_further_ticker
			every_country = {
				limit = {
					has_country_flag = empire_of_china_nomadic_border_cause
				}
				clr_country_flag = empire_of_china_nomadic_border_cause
			}
		}
	}

	#triggered monthly when it is ticking up to 100. The progress is saved in a variable that is <disaster key>_progress
	on_progress_effect = {
		if = {
			limit = {
				NOT = { has_country_flag = empire_of_china_nomadic_border_started_ticker }
				check_variable = {
					which = empire_of_china_nomadic_border_progress
					value = 10
				}
			}
			set_country_flag = empire_of_china_nomadic_border_started_ticker
			every_neighbor_country = {
				limit = {
					OR = {
						has_reform = steppe_horde
						has_reform = great_mongol_state_reform
					}
					total_own_and_non_tributary_subject_development = 300
					NOT = { alliance_with = ROOT }
					is_subject = no
				}
				set_country_flag = empire_of_china_nomadic_border_cause
				country_event = { id = empire_of_china_nomadic_border_events.3 }
			}
		}
		if = {
			limit = {
				NOT = { has_country_flag = empire_of_china_nomadic_border_further_ticker }
				check_variable = {
					which = empire_of_china_nomadic_border_progress
					value = 75
				}
			}
			set_country_flag = empire_of_china_nomadic_border_further_ticker
			every_neighbor_country = {
				limit = {
					OR = {
						has_reform = steppe_horde
						has_reform = great_mongol_state_reform
					}
					total_own_and_non_tributary_subject_development = 300
					NOT = { alliance_with = ROOT }
					is_subject = no
				}
				country_event = { id = empire_of_china_nomadic_border_events.4 }
			}
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						which = empire_of_china_nomadic_border_progress
						value = 5
					}
				}
				has_country_flag = empire_of_china_nomadic_border_started_ticker
			}

			every_country = {
				limit = {
					has_country_flag = empire_of_china_nomadic_border_cause
				}
				country_event = { id = empire_of_china_nomadic_border_events.5 }
			}
		}
	}
}
