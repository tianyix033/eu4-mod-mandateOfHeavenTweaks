namespace = mht_nomadic

###############################################################################
# Nomadic Border disaster - the standoff resolves one way or the other.
#
# Vanilla lets the Emperor of China sit at zero mandate indefinitely: the
# disaster never ends, and because a country can only have one disaster at a
# time, the Ming Crisis can never fire either. These events force the question.
#
#   mht_nomadic.1  hidden picker, fires on the Emperor from the disaster's
#                  on_monthly, selects the horde and passes it the ultimatum
#   mht_nomadic.2  the ultimatum, fires on the horde        (FROM = Emperor)
#   mht_nomadic.3  the Emperor's answer, fires on the Emperor (FROM = horde)
###############################################################################


# ------------------------------------- the court resolves to act, on EoC ----
# Fires on the Emperor from the disaster's on_monthly once mandate has fallen
# below 20. The horde is chosen in immediate so the description can name it;
# the option carries the ultimatum to them.
#
# EU4 has no scope sorting, so "the largest horde" is approximated by trying
# the biggest development band first and falling back. A horde reaching 300
# development on China's border is rare enough that there is usually only one
# candidate anyway.
country_event = {
	id = mht_nomadic.1
	title = mht_nomadic.1.t
	desc = mht_nomadic.1.d
	picture = CHINESE_GENERALS_eventPicture

	is_triggered_only = yes

	trigger = {
		is_emperor_of_china = yes
		has_disaster = empire_of_china_nomadic_border
		NOT = { imperial_mandate = 20 }
		NOT = { has_country_flag = mht_nomadic_ultimatum_sent }
		any_neighbor_country = {
			is_subject = no
			NOT = { truce_with = ROOT }
			NOT = { alliance_with = ROOT }
			OR = {
				has_reform = steppe_horde
				has_reform = great_mongol_state_reform
			}
			total_own_and_non_tributary_subject_development = 300
		}
	}

	immediate = {
		hidden_effect = {
			set_country_flag = mht_nomadic_ultimatum_sent
			save_event_target_as = mht_emperor
			if = {
				limit = {
					any_neighbor_country = {
						is_subject = no
						NOT = { truce_with = ROOT }
						NOT = { alliance_with = ROOT }
						OR = { has_reform = steppe_horde  has_reform = great_mongol_state_reform }
						total_own_and_non_tributary_subject_development = 700
					}
				}
				random_neighbor_country = {
					limit = {
						is_subject = no
						NOT = { truce_with = ROOT }
						NOT = { alliance_with = ROOT }
						OR = { has_reform = steppe_horde  has_reform = great_mongol_state_reform }
						total_own_and_non_tributary_subject_development = 700
					}
					save_event_target_as = mht_horde
				}
			}
			else_if = {
				limit = {
					any_neighbor_country = {
						is_subject = no
						NOT = { truce_with = ROOT }
						NOT = { alliance_with = ROOT }
						OR = { has_reform = steppe_horde  has_reform = great_mongol_state_reform }
						total_own_and_non_tributary_subject_development = 500
					}
				}
				random_neighbor_country = {
					limit = {
						is_subject = no
						NOT = { truce_with = ROOT }
						NOT = { alliance_with = ROOT }
						OR = { has_reform = steppe_horde  has_reform = great_mongol_state_reform }
						total_own_and_non_tributary_subject_development = 500
					}
					save_event_target_as = mht_horde
				}
			}
			else = {
				random_neighbor_country = {
					limit = {
						is_subject = no
						NOT = { truce_with = ROOT }
						NOT = { alliance_with = ROOT }
						OR = { has_reform = steppe_horde  has_reform = great_mongol_state_reform }
						total_own_and_non_tributary_subject_development = 300
					}
					save_event_target_as = mht_horde
				}
			}
		}
	}

	option = {
		name = mht_nomadic.1.a
		event_target:mht_horde = { country_event = { id = mht_nomadic.2 } }
	}
}


# ------------------------------------------- the ultimatum, on the horde ----
# ROOT = the horde, FROM = the Emperor of China
country_event = {
	id = mht_nomadic.2
	title = mht_nomadic.2.t
	desc = mht_nomadic.2.d
	picture = HORDE_eventPicture

	is_triggered_only = yes

	# invade now
	option = {
		name = mht_nomadic.2.a
		ai_chance = {
			factor = 60
			modifier = {					#@ stronger than the Emperor - the moment is now
				factor = 6
				army_strength = {
					who = FROM
					value = 0.9			#@ 0.9 not 1.0: army_strength misses the
				}								#@ Emperor's morale and discipline losses
			}
			modifier = {					#@ badly outmatched. army_strength weighs size and mil tech
				factor = 0					#@ but not morale or discipline, which a low-mandate
				NOT = { army_strength = { who = FROM  value = 0.5 } }						#@ Emperor lacks, so this threshold favours the horde
			}
			modifier = {					#@ bankrupt - no war is affordable
				factor = 0
				is_bankrupt = yes
			}
			modifier = {					#@ no men to send
				factor = 0
				NOT = { manpower_percentage = 0.33 }
			}
			modifier = {					#@ already losing a war elsewhere
				factor = 0
				is_at_war = yes
				NOT = { war_score = -25 }
			}
			modifier = {					#@ at war but holding - still a poor moment for a second front
				factor = 0.25
				is_at_war = yes
				war_score = -25
			}
		}
		add_horde_unity = 10
		add_army_tradition = 5
		declare_war_with_cb = {
			who = FROM
			casus_belli = cb_horde_vs_civ
		}
	}

	# submit
	option = {
		name = mht_nomadic.2.b
		ai_chance = {
			factor = 2
			modifier = {					#@ stronger than the Emperor - submission is unthinkable
				factor = 0.1
				army_strength = {
					who = FROM
					value = 0.9			#@ 0.9 not 1.0: army_strength misses the
				}								#@ Emperor's morale and discipline losses
			}
			modifier = {					#@ badly outmatched
				factor = 25
				NOT = { army_strength = { who = FROM  value = 0.5 } }
			}
			modifier = {					#@ bankrupt - submission is all that is left
				factor = 1000
				is_bankrupt = yes
			}
			modifier = {					#@ no men to send
				factor = 10
				NOT = { manpower_percentage = 0.33 }
			}
			modifier = {					#@ already losing a war - cannot afford another enemy
				factor = 60
				is_at_war = yes
				NOT = { war_score = -25 }
			}
		}
		add_horde_unity = -20
		add_prestige = -10
		FROM = {
			create_subject = {
				subject_type = tributary_state
				subject = ROOT
			}
		}
	}

	# refuse, and let them come
	option = {
		name = mht_nomadic.2.c
		ai_chance = {
			factor = 40
			modifier = {					#@ outmatched, but defiance costs nothing today
				factor = 1.25
				NOT = { army_strength = { who = FROM  value = 0.5 } }
			}
			modifier = {					#@ bankrupt - defiance is not affordable either
				factor = 0
				is_bankrupt = yes
			}
			modifier = {					#@ no men to send
				factor = 1.5
				NOT = { manpower_percentage = 0.33 }
			}
		}
		add_horde_unity = 5
		FROM = { country_event = { id = mht_nomadic.3 } }
	}
}


# -------------------------------------- the Emperor's answer, on the EoC ----

	desc = mht_nomadic.3.d
	picture = CHINESE_GENERALS_eventPicture

	# back down
	option = {
		name = mht_nomadic.3.a

			factor = 1

		add_estate_loyalty = { estate = estate_nobles   loyalty = -10 }
		add_estate_loyalty = { estate = estate_church   loyalty = -10 }
	}

	# show superiority - insult wins on warscore, no desert sieges needed
	option = {
		name = mht_nomadic.3.b

			factor = 200					#@ short of men and not clearly dominant: win on warscore
			modifier = {					#@ rather than besiege the steppe
				factor = 0

					AND = {
				NOT = { manpower_percentage = 0.33 }
				NOT = { army_strength = { who = FROM  value = 1.5 } }
					}
				}

			modifier = {					#@ bankrupt
				factor = 0

			}

				factor = 0
				NOT = { army_strength = { who = FROM  value = 1.0 } }
			}
			modifier = {					#@ already losing a war badly
				factor = 0

				NOT = { war_score = -25 }
			}
			modifier = {					#@ unification outranks this
				factor = 0
				has_casus_belli = { type = cb_chinese_unification  target = FROM }
			}
			modifier = {					#@ imperialism outranks this
				factor = 0

			}
		}
		declare_war_with_cb = {

				factor = 0

				NOT = { manpower_percentage = 0.33 }
				NOT = { army_strength = { who = FROM  value = 1.5 } }
				}
			}
			modifier = {					#@ bankrupt
				factor = 0
				is_bankrupt = yes
			}
			modifier = {					#@ not actually stronger than them
				factor = 0
				NOT = { army_strength = { who = FROM  value = 1.0 } }
			}
			modifier = {					#@ already losing a war badly
				factor = 0

				NOT = { war_score = -25 }

				factor = 0
				has_casus_belli = { type = cb_chinese_unification  target = FROM }
			}
			modifier = {					#@ imperialism outranks this
				factor = 0

			}

	# imperialism
	option = {

# ------------------------------------------- disaster over, reset the flag ----
# The disaster's on_end takes a single event, and vanilla already points it at
# empire_of_china_nomadic_border_events.2. This stands in front of that event,
# clears our once-only flag so the standoff can happen again if the disaster
# recurs, then hands straight over so vanilla's +30 mandate and +2 stability
# still land.
country_event = {
	id = mht_nomadic.4
	title = mht_nomadic.4.t
	desc = mht_nomadic.4.d
	picture = CHINESE_GENERAL_eventPicture

	hidden = yes
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			clr_country_flag = mht_nomadic_ultimatum_sent
			#@ Two ways out of this disaster now, and they are not worth the same.
			#@
			#@ A truce ends it while the horde is still an independent 300+ dev
			#@ horde on our border - nothing was actually settled, we just bought
			#@ years. Vanilla's end event pays +30 mandate and +2 stability, which
			#@ on a 5-year truce cycle is farmable, and it would pay out for
			#@ LOSING the war just as readily.
			#@
			#@ So: if a shielding truce is what cleared it, take the weaker
			#@ mht_nomadic.5 instead. Every other route - the horde subjugated,
			#@ driven under 300 dev, allied, de-horded, or our losing the mandate
			#@ outright - falls through to vanilla's event unchanged.
			#@
			#@ The conditions below deliberately mirror the disaster's own
			#@ qualifying test in common/disasters/. A tributary horde fails
			#@ is_subject = no and a stripped one fails the development test, so
			#@ the force-tributary and unification wars still pay full price even
			#@ though those peaces set a truce too.
			if = {
				limit = {
					is_emperor_of_china = yes
					any_neighbor_country = {
						truce_with = ROOT
						is_subject = no
						NOT = { alliance_with = ROOT }
						OR = {
							has_reform = steppe_horde
							has_reform = great_mongol_state_reform
						}
						total_own_and_non_tributary_subject_development = 300
					}
				}
				country_event = { id = mht_nomadic.5 }
			}
			else = {
				country_event = { id = empire_of_china_nomadic_border_events.2 }
			}
		}
	}

	option = {
		name = mht_nomadic.4.a
	}
}

# The Border Quiets
#
# The weaker end of the disaster, for when a truce cleared it rather than a
# settlement. Deliberately says nothing about who won: can_end fires on any
# peace, so this same event covers a victory and a defeat.
#
# on_non_estate_disaster_end is vanilla's own bookkeeping from
# empire_of_china_nomadic_border_events.2 - it increments
# total_disasters_completed_counter. Kept here so the counter stays correct on
# this path too.
########################################
country_event = {
	id = mht_nomadic.5
	title = mht_nomadic.5.t
	desc = mht_nomadic.5.d
	picture = CHINESE_GENERAL_eventPicture

	is_triggered_only = yes
	major = yes

	immediate = {
		on_non_estate_disaster_end = yes
	}

	option = {
		name = mht_nomadic.5.a
		add_mandate = 10			#@ vanilla's end event pays 30
		add_stability = 1			#@ vanilla's end event pays 2
	}
}
